    </div>

    <script>
        async function getCodeResponse() {
            const codeInput = document.getElementById('codeInput').value;
            const codeResponseElement = document.getElementById('codeResponse');

            // Replace 'YOUR_OPENAI_API_KEY' with your actual OpenAI API key
            const apiKey = 'YOUR_OPENAI_API_KEY';

            const response = await fetch('/get_code_response', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Authorization': `Bearer ${apiKey}`
                },
                body: JSON.stringify({ prompt: codeInput })
            });

            const responseData = await response.json();
            codeResponseElement.innerHTML = `<p>${responseData.response}</p>`;
        }
    </script>
