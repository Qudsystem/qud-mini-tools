Here is a simple example using JavaScript to translate a text from English to Arabic using the Google Translate API. Note that this is just a basic example, and you may need to adjust it based on your specific requirements and the translation service you're using.

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Text Translation</title>
</head>
<body>

<div id="originalText">
    <p>Hi, I'm Rahul. How can I help you today?</p>
</div>

<div id="translatedText"></div>

<script>
    const originalText = document.getElementById('originalText').innerText;
    const apiKey = 'YOUR_GOOGLE_TRANSLATE_API_KEY'; // Replace with your API key

    // Use Google Translate API
    fetch(`https://translation.googleapis.com/language/translate/v2?key=${apiKey}`, {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
        },
        body: JSON.stringify({
            q: originalText,
            target: 'ar', // Arabic language code
        }),
    })
    .then(response => response.json())
    .then(data => {
        const translatedText = data.data.translations[0].translatedText;
        document.getElementById('translatedText').innerText = translatedText;
    })
    .catch(error => console.error('Error:', error));
</script>

</body>
</html>

Remember to replace 'YOUR_GOOGLE_TRANSLATE_API_KEY' with your actual Google Translate API key. Also, make sure to check the terms of use for the translation service you're using.
